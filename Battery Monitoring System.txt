/************************************************************
 *                 BATTERY PARAMETERS
 ************************************************************/

// 3S Li-ion battery
const float BATTERY_FULL_VOLTAGE  = 12.60;
const float BATTERY_EMPTY_VOLTAGE = 9.60;

// Rated battery capacity
const float RATED_CAPACITY_AH = 3.0;

/************************************************************
 *                 VOLTAGE SENSOR
 ************************************************************/

// Voltage divider:
// Battery + ---- R1 ---- ADC ---- R2 ---- GND

const float R1 = 100000.0;
const float R2 = 27000.0;

// Calibration multiplier.
// Adjust this after comparing with a multimeter.
float voltageCalibration = 1.000;

/************************************************************
 *                 ACS712 5A
 ************************************************************/

// ACS712 5A sensitivity
// Approximately 185 mV/A

const float ACS712_SENSITIVITY = 0.185;

// Nominal zero-current output.
// Normally around 2.5 V.
// Calibrate this value with no load.
float ACS712_ZERO_VOLTAGE = 2.500;

/************************************************************
 *                 ADC SETTINGS
 ************************************************************/

const float ADC_REFERENCE = 3.3;
const int ADC_MAX_VALUE = 4095;

/************************************************************
 *                 SYSTEM THRESHOLDS
 ************************************************************/

// Below/equal this voltage, battery parameters are zero.
const float SYSTEM_MIN_VOLTAGE = 2.0;

// User requested current = zero below 1 V.
const float CURRENT_CUTOFF_VOLTAGE = 1.0;

/************************************************************
 *                 TEMPERATURE LIMITS
 ************************************************************/

const float FAN_ON_TEMP     = 35.0;
const float PELTIER_ON_TEMP = 50.0;

/************************************************************
 *                 SYSTEM VARIABLES
 ************************************************************/

float batteryVoltage = 0.0;
float current_A      = 0.0;
float temperatureC   = 0.0;
float power_W        = 0.0;

float batterySOC = 0.0;
float batterySOH = 0.0;
float batteryRUL = 0.0;

int cycleNumber = 0;

bool fanStatus = false;
bool peltierStatus = false;

unsigned long lastDisplayUpdate = 0;
unsigned long lastSensorUpdate = 0;

/************************************************************
 *                 LCD PAGE
 ************************************************************/

int lcdPage = 0;

/************************************************************
 *                 ADC AVERAGING
 ************************************************************/

float readADCVoltage(int pin, int samples = 20)
{
  uint32_t total = 0;

  for (int i = 0; i < samples; i++)
  {
    total += analogRead(pin);
    delayMicroseconds(300);
  }

  float averageADC = (float)total / samples;

  return (averageADC / ADC_MAX_VALUE) * ADC_REFERENCE;
}

/************************************************************
 *                 BATTERY VOLTAGE
 ************************************************************/

float readBatteryVoltage()
{
  float adcVoltage = readADCVoltage(VOLTAGE_PIN, 25);

  float batteryVoltageCalculated =
      adcVoltage * ((R1 + R2) / R2);

  batteryVoltageCalculated *= voltageCalibration;

  return batteryVoltageCalculated;
}

/************************************************************
 *                 ACS712 CURRENT
 ************************************************************/

float readCurrent()
{
  // If battery voltage is too low,
  // current must be zero.
  if (batteryVoltage < CURRENT_CUTOFF_VOLTAGE)
  {
    return 0.0;
  }

  float sensorVoltage = readADCVoltage(CURRENT_PIN, 50);

  float current =
      (sensorVoltage - ACS712_ZERO_VOLTAGE)
      / ACS712_SENSITIVITY;

  /*
   * Small ADC/noise variations can create
   * tiny positive/negative values.
   */

  if (fabs(current) < 0.05)
  {
    current = 0.0;
  }

  // For battery load measurement, negative current
  // is normally not required.
  if (current < 0.0)
  {
    current = 0.0;
  }

  /*
   * ACS712 5A module is normally used around
   * the 0-5A range.
   */
  if (current > 5.0)
  {
    current = 5.0;
  }

  return current;
}

/************************************************************
 *                 TEMPERATURE
 ************************************************************/

float readTemperature()
{
  temperatureSensor.requestTemperatures();

  float temp =
      temperatureSensor.getTempCByIndex(0);

  // DS18B20 disconnected/error
  if (temp == DEVICE_DISCONNECTED_C ||
      temp < -55.0 ||
      temp > 125.0)
  {
    return NAN;
  }

  return temp;
}

/************************************************************
 *                 BATTERY SOC
 ************************************************************/

float calculateSOC(float voltage)
{
  if (voltage <= SYSTEM_MIN_VOLTAGE)
  {
    return 0.0;
  }

  if (voltage <= BATTERY_EMPTY_VOLTAGE)
  {
    return 0.0;
  }

  if (voltage >= BATTERY_FULL_VOLTAGE)
  {
    return 100.0;
  }

  float soc =
      ((voltage - BATTERY_EMPTY_VOLTAGE) /
       (BATTERY_FULL_VOLTAGE - BATTERY_EMPTY_VOLTAGE))
      * 100.0;

  return constrain(soc, 0.0, 100.0);
}

/************************************************************
 *                 DEMO SOH MODEL
 ************************************************************/

float calculateSOH()
{
  if (batteryVoltage <= SYSTEM_MIN_VOLTAGE)
  {
    return 0.0;
  }

  /*
   * Educational demonstration model.
   *
   * The value changes gradually rather than being
   * completely random every second.
   */

  static float simulatedSOH = 96.0;

  static unsigned long lastChange = 0;

  if (millis() - lastChange >= 15000)
  {
    lastChange = millis();

    float change =
        random(-10, 11) / 10.0;

    simulatedSOH += change;

    simulatedSOH =
        constrain(simulatedSOH, 70.0, 100.0);
  }

  return simulatedSOH;
}

/************************************************************
 *                 DEMO CYCLE NUMBER
 ************************************************************/

int calculateCycleNumber()
{
  if (batteryVoltage <= SYSTEM_MIN_VOLTAGE)
  {
    return 0;
  }

  static int simulatedCycle = 150;

  static unsigned long lastCycleChange = 0;

  if (millis() - lastCycleChange >= 30000)
  {
    lastCycleChange = millis();

    simulatedCycle += random(0, 3);

    if (simulatedCycle > 1000)
    {
      simulatedCycle = 1000;
    }
  }

  return simulatedCycle;
}

/************************************************************
 *                 RUL CALCULATION
 ************************************************************/

float calculateRUL(float soh)
{
  if (batteryVoltage <= SYSTEM_MIN_VOLTAGE)
  {
    return 0.0;
  }

  if (soh <= 0.0)
  {
    return 0.0;
  }

  /*
   * Demonstration:
   *
   * Assume battery EOL = 80% SOH.
   * Remaining useful life is represented
   * as an estimated number of cycles.
   */

  const float END_OF_LIFE_SOH = 80.0;

  const float ESTIMATED_REMAINING_CYCLES = 500.0;

  if (soh <= END_OF_LIFE_SOH)
  {
    return 0.0;
  }

  float rul =
      ((soh - END_OF_LIFE_SOH) /
       (100.0 - END_OF_LIFE_SOH))
      * ESTIMATED_REMAINING_CYCLES;

  return constrain(rul, 0.0, ESTIMATED_REMAINING_CYCLES);
}

/************************************************************
 *                 COOLING CONTROL
 ************************************************************/

void updateCooling()
{
  if (isnan(temperatureC))
  {
    /*
     * Temperature sensor failure:
     * turn cooling devices ON as a safety condition.
     */
    fanStatus = true;
    peltierStatus = true;
  }
  else
  {
    /*
     * <35°C
     * FAN OFF
     * PELTIER OFF
     */
    if (temperatureC < FAN_ON_TEMP)
    {
      fanStatus = false;
      peltierStatus = false;
    }

    /*
     * 35°C to 50°C
     * FAN ON
     * PELTIER OFF
     */
    else if (temperatureC <= PELTIER_ON_TEMP)
    {
      fanStatus = true;
      peltierStatus = false;
    }

    /*
     * >50°C
     * FAN ON
     * PELTIER ON
     */
    else
    {
      fanStatus = true;
      peltierStatus = true;
    }
  }

  digitalWrite(FAN_RELAY_PIN,
               fanStatus ? HIGH : LOW);

  digitalWrite(PELTIER_RELAY_PIN,
               peltierStatus ? HIGH : LOW);
}

/************************************************************
 *                 BUZZER CONTROL
 ************************************************************/

void updateBuzzer()
{
  /*
   * High temperature warning.
   */

  if (!isnan(temperatureC) &&
      temperatureC > PELTIER_ON_TEMP)
  {
    digitalWrite(BUZZER_PIN, HIGH);
  }
  else
  {
    digitalWrite(BUZZER_PIN, LOW);
  }
}

/************************************************************
 *                 SENSOR PROCESSING
 ************************************************************/

void readAllSensors()
{
  // 1. REAL battery voltage
  batteryVoltage = readBatteryVoltage();

  // 2. REAL DS18B20 temperature
  temperatureC = readTemperature();

  /*
   * Required logic:
   *
   * Voltage <= 2V:
   * Current = 0
   * Power = 0
   * SOC = 0
   * SOH = 0
   * RUL = 0
   * Cycle = 0
   *
   * Temperature remains REAL.
   */

  if (batteryVoltage <= SYSTEM_MIN_VOLTAGE)
  {
    current_A = 0.0;
    power_W = 0.0;

    batterySOC = 0.0;
    batterySOH = 0.0;
    batteryRUL = 0.0;

    cycleNumber = 0;
  }

  else
  {
    // REAL ACS712 current
    current_A = readCurrent();

    // Electrical power
    power_W =
        batteryVoltage * current_A;

    // SOC from actual battery voltage
    batterySOC =
        calculateSOC(batteryVoltage);

    /*
     * Demonstration values.
     * Replace with a trained battery degradation
     * model for a real RUL product.
     */
    batterySOH =
        calculateSOH();

    cycleNumber =
        calculateCycleNumber();

    batteryRUL =
        calculateRUL(batterySOH);
  }

  // Cooling is ALWAYS controlled using
  // the REAL DS18B20 temperature.
  updateCooling();

  updateBuzzer();
}

/************************************************************
 *                 LCD DISPLAY
 ************************************************************/

void displayLCD()
{
  lcd.clear();

  if (lcdPage == 0)
  {
    lcd.setCursor(0, 0);
    lcd.print("V:");
    lcd.print(batteryVoltage, 2);
    lcd.print("V ");

    lcd.print("I:");
    lcd.print(current_A, 1);

    lcd.setCursor(0, 1);
    lcd.print("SOC:");
    lcd.print(batterySOC, 0);
    lcd.print("% ");

    lcd.print("T:");

    if (isnan(temperatureC))
      lcd.print("ERR");
    else
      lcd.print(temperatureC, 1);
  }

  else if (lcdPage == 1)
  {
    lcd.setCursor(0, 0);
    lcd.print("SOH:");
    lcd.print(batterySOH, 1);
    lcd.print("%");

    lcd.setCursor(0, 1);
    lcd.print("RUL:");
    lcd.print(batteryRUL, 0);
    lcd.print(" Cy");
  }

  else if (lcdPage == 2)
  {
    lcd.setCursor(0, 0);
    lcd.print("Fan:");

    if (fanStatus)
      lcd.print("ON ");
    else
      lcd.print("OFF");

    lcd.print(" P:");

    if (peltierStatus)
      lcd.print("ON");
    else
      lcd.print("OFF");

    lcd.setCursor(0, 1);

    if (isnan(temperatureC))
      lcd.print("TEMP SENSOR ERR");
    else
    {
      lcd.print("Temp:");
      lcd.print(temperatureC, 1);
      lcd.print("C");
    }
  }

  lcdPage++;

  if (lcdPage > 2)
  {
    lcdPage = 0;
  }
}

/************************************************************
 *                 BLYNK UPDATE
 ************************************************************/

void updateBlynk()
{
  /*
   * Blynk Virtual Pins
   *
   * V0  Battery Voltage
   * V1  Current
   * V2  Temperature
   * V3  Power
   * V4  SOC
   * V5  SOH
   * V6  RUL
   * V7  Cooling/Fan
   * V8  Peltier
   * V9  Cycle Number
   */

  Blynk.virtualWrite(V0, batteryVoltage);
  Blynk.virtualWrite(V1, current_A);

  if (isnan(temperatureC))
    Blynk.virtualWrite(V2, 0);
  else
    Blynk.virtualWrite(V2, temperatureC);

  Blynk.virtualWrite(V3, power_W);
  Blynk.virtualWrite(V4, batterySOC);
  Blynk.virtualWrite(V5, batterySOH);
  Blynk.virtualWrite(V6, batteryRUL);

  Blynk.virtualWrite(V7,
                     fanStatus ? 1 : 0);

  Blynk.virtualWrite(V8,
                     peltierStatus ? 1 : 0);

  Blynk.virtualWrite(V9, cycleNumber);
}

void sensorTask()

void lcdTask()

  displayLCD();
}

/************************************************************
 *                 SETUP
 ************************************************************/

void setup()
{
  Serial.begin(115200);

  delay(1000);

  Serial.println();
  Serial.println("======================================");
  Serial.println(" BATTERY RUL ESTIMATOR");
  Serial.println(" ESP32 + Blynk IoT");
  Serial.println("======================================");

  /******************************************************
   * GPIO CONFIGURATION
   ******************************************************/

  pinMode(VOLTAGE_PIN, INPUT);
  pinMode(CURRENT_PIN, INPUT);

  pinMode(BUZZER_PIN, OUTPUT);
  pinMode(FAN_RELAY_PIN, OUTPUT);
  pinMode(PELTIER_RELAY_PIN, OUTPUT);

  digitalWrite(BUZZER_PIN, LOW);
  digitalWrite(FAN_RELAY_PIN, LOW);
  digitalWrite(PELTIER_RELAY_PIN, LOW);

  /******************************************************
   * ADC CONFIGURATION
   ******************************************************/

  analogReadResolution(12);

  analogSetPinAttenuation(
      VOLTAGE_PIN,
      ADC_11db);

  analogSetPinAttenuation(
      CURRENT_PIN,
      ADC_11db);

  /******************************************************
   * I2C
   ******************************************************/

  Wire.begin(21, 22);

  /******************************************************
   * LCD
   ******************************************************/

  lcd.init();
  lcd.backlight();

  lcd.clear();

  lcd.setCursor(0, 0);
  lcd.print("Battery RUL");

  lcd.setCursor(0, 1);
  lcd.print("Initializing...");

  delay(2000);

  /******************************************************
   * DS18B20
   ******************************************************/

  temperatureSensor.begin();

  temperatureSensor.setResolution(12);

  /******************************************************
   * RANDOM SEED
   ******************************************************/

  randomSeed(
      analogRead(VOLTAGE_PIN) +
      micros());

  /******************************************************
   * BLYNK
   ******************************************************/

  Blynk.begin(
      BLYNK_AUTH_TOKEN,
      ssid,
      pass);

  lcd.clear();

  lcd.setCursor(0, 0);
  lcd.print("System Ready");

  delay(1000);

  /******************************************************
   * TIMERS
   ******************************************************/

  // Sensor reading every 2 seconds
  timer.setInterval(
      2000L,
      sensorTask);

  // LCD update every 3 seconds
  timer.setInterval(
      3000L,
      lcdTask);

  // Blynk update every 2 seconds
  timer.setInterval(
      2000L,
      updateBlynk);

  /******************************************************
   * INITIAL SENSOR READING
   ******************************************************/

  readAllSensors();

  displayLCD();

  Serial.println("System initialization completed.");
}

/************************************************************
 *                 MAIN LOOP
 ************************************************************/

void loop()
{
  Blynk.run();

  timer.run();
}
